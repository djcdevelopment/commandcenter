054| def _grep(text: str, pattern: str, group: int = 1, cast: Callable = str, default=None):
055|     m = re.search(pattern, text, re.MULTILINE)

792| def render_into_doc(rows: list[Row]) -> bool:
793|     text = DOC.read_text(encoding="utf-8") if DOC.exists() else f"# Sizing map\n\n{BEGIN}\n{END}\n"
